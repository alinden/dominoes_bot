cmake_minimum_required(VERSION 3.16)
project(dominoes_bot LANGUAGES CXX)

add_library(dominoes_engine src/board.cpp src/domino.cpp src/game.cpp)
target_include_directories(dominoes_engine PUBLIC include)
target_compile_features(dominoes_engine PUBLIC cxx_std_20)
set_target_properties(dominoes_engine PROPERTIES CXX_EXTENSIONS OFF)

if(MSVC)
    target_compile_options(dominoes_engine PRIVATE /W4)
else()
    target_compile_options(dominoes_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_library(dominoes_greedy src/greedy.cpp)
target_link_libraries(dominoes_greedy PUBLIC dominoes_engine)
add_executable(dominoes src/terminal.cpp)
target_link_libraries(dominoes PRIVATE dominoes_greedy)
set_target_properties(dominoes_greedy dominoes PROPERTIES CXX_EXTENSIONS OFF)

include(CTest)
if(BUILD_TESTING)
    add_executable(domino_tests tests/domino_tests.cpp)
    target_link_libraries(domino_tests PRIVATE dominoes_engine)
    set_target_properties(domino_tests PROPERTIES CXX_EXTENSIONS OFF)
    add_test(NAME domino_tests COMMAND domino_tests)
    add_executable(board_tests tests/board_tests.cpp)
    target_link_libraries(board_tests PRIVATE dominoes_engine)
    set_target_properties(board_tests PROPERTIES CXX_EXTENSIONS OFF)
    add_test(NAME board_tests COMMAND board_tests)
    add_executable(game_tests tests/game_tests.cpp)
    target_link_libraries(game_tests PRIVATE dominoes_engine)
    set_target_properties(game_tests PROPERTIES CXX_EXTENSIONS OFF)
    add_test(NAME game_tests COMMAND game_tests)
    add_executable(scala_replay_tests tests/scala_replay_tests.cpp)
    target_link_libraries(scala_replay_tests PRIVATE dominoes_engine)
    set_target_properties(scala_replay_tests PROPERTIES CXX_EXTENSIONS OFF)
    add_test(NAME scala_replay_tests
        COMMAND scala_replay_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/scala_replay.txt")
endif()
